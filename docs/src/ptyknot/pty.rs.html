6//! Low-level pseudo-tty setup routines.
7//!
8//! This module contains several pty setup functions and a
9//! "special" `waitpid()` implementation.
10//!
11//! Much of this is code borrowed from <http://github.com/stemjail/tty-rs>.
12
13use std::ffi::{CStr, OsString};
14use std::fs::File;
15use std::io::{Error, ErrorKind, Result};
16use std::os::unix::ffi::OsStringExt;
17use std::os::unix::io::{AsRawFd, FromRawFd, RawFd};
18use std::path::*;
19
20use libc::{c_int, pid_t};
21
22mod raw {
23    use libc::{c_char, c_int};
24    pub use libc::{close, dup2, pipe, waitpid};
25
26    unsafe extern "C" {
27        pub fn grantpt(fd: c_int) -> c_int;
28        pub fn unlockpt(fd: c_int) -> c_int;
29        pub fn ptsname(fd: c_int) -> *const c_char;
30    }
31}
32
33/// Change the mode and owner of the slave pty associated
34/// with a given open master. See `grantpt(3)` in the UNIX
35/// manual pages for details.
36pub fn grantpt(master: &File) -> Result<()> {
37    // # Safety
38    // `grantpt()` is passed a valid file descriptor.
39    match unsafe { raw::grantpt(master.as_raw_fd()) } {
40        -1 => Err(Error::last_os_error()),
41        _ => Ok(()),
42    }
43}
44
45/// "Unlocks" the slave pty associated with the give nopen
46/// master. See `unlockpt(3)` in the UNIX manual pages for
47/// details.
48pub fn unlockpt(master: &File) -> Result<()> {
49    // # Safety
50    // `unlockpt()` will safely accept an invalid file descriptor.
51    match unsafe { raw::unlockpt(master.as_raw_fd()) } {
52        -1 => Err(Error::last_os_error()),
53        _ => Ok(()),
54    }
55}
56
57/// Returns the name of the slave pty associated with the
58/// given open master. See `unlockpt(3)` in the UNIX manual
59/// pages for details.
60pub fn ptsname(master: &File) -> Result<PathBuf> {
61    // # Safety
62    // `ptsname()` is passed a valid file descriptor.
63    let cstr = match unsafe { raw::ptsname(master.as_raw_fd()).as_ref() } {
64        None => return Err(Error::last_os_error()),
65        // # Safety
66        // `ptsname()` is guaranteed to return a valid pointer
67        // when it does not return an error.
68        Some(ptr) => unsafe { CStr::from_ptr(ptr) },
69    };
70    let buf = match cstr.to_str() {
71        Ok(s) => s,
72        Err(e) => return Err(Error::new(ErrorKind::InvalidData, e)),
73    };
74    let os_string = OsString::from_vec(buf.as_bytes().to_vec());
75    Ok(PathBuf::from(os_string))
76}

78/// Blocking wait until process completes. Returns the
79/// process exit status. See `waitpid(3)` in the UNIX manual
80/// pages for details.
81pub fn waitpid(pid: i32) -> Result<i32> {
82    let mut status: c_int = 0;
83    // # Safety
84    // `waitpid()` will safely accept an invalid `pid`.
85    match unsafe { raw::waitpid(pid as pid_t, &mut status as *mut c_int, 0) } {
86        -1 => Err(Error::last_os_error()),
87        _ => Ok(status),
88    }
89}

91/// Close a raw file descriptor in the child process.
92pub(crate) fn close(fd: RawFd) -> Result<()> {
93    // # Safety
94    // `close()` accepts any raw file descriptor value.
95    match unsafe { raw::close(fd) } {
96        -1 => Err(Error::last_os_error()),
97        _ => Ok(()),
98    }
99}

101/// Make the underlying file of `dst` refer to the
102/// underlying file of `src`. If `dst` is open, it will
103/// be closed first. See `dup2(2)` in the UNIX manual
104/// pages for details.
105pub fn dup2(old: &File, new_fd: crate::StdFd) -> Result<()> {
106    // # Safety
107    // `dup2()` will safely accept invalid file descriptors.
108    match unsafe { raw::dup2(old.as_raw_fd(), new_fd.as_raw_fd()) } {
109        -1 => Err(Error::last_os_error()),
110        _ => Ok(()),
111    }
112}

114/// Make a new pipe. The 0-th side of the resulting array is
115/// the read side. The 1-th side is the write side. See
116/// `pipe(2)` in the UNIX manual pages for details.
117pub fn pipe() -> Result<[File; 2]> {
118    let mut pipefds: [RawFd; 2] = [0; 2];
119    // # Safety
120    // `pipe()` mutates only array to be returned, supplying
121    // it with valid values before returning success.
122    let result = unsafe { raw::pipe(pipefds.as_mut_ptr()) };
123    if result == -1 {
124        return Err(Error::last_os_error());
125    }
126    Ok(std::array::from_fn(|i| {
127        // # Safety
128        // On success `pipefds` will contain valid fds.
129        unsafe { FromRawFd::from_raw_fd(pipefds[i]) }
130    }))
