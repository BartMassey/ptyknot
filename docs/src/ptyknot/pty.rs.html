6//! Low-level pseudo-tty setup routines.
7//!
8//! This module contains several pty setup functions and a
9//! "special" `waitpid()` implementation.
10//!
11//! Much of this is code borrowed from <http://github.com/stemjail/tty-rs>.
12

14use std::fs::File;
15use std::io::{Error, ErrorKind, Result};
16use std::os::unix::ffi::OsStringExt;
17use std::os::unix::io::{AsRawFd, FromRawFd, RawFd};

22use libc::{c_int, pid_t};

24mod raw {
25    #[cfg(test)]
26    pub use libc::close;
27    use libc::{c_char, c_int};
28    pub use libc::{dup2, pipe, waitpid};

30    unsafe extern "C" {
31        pub fn grantpt(fd: c_int) -> c_int;
32        pub fn unlockpt(fd: c_int) -> c_int;
33        pub fn ptsname(fd: c_int) -> *const c_char;
34    }

37/// Change the mode and owner of the slave pty associated
38/// with a given open master. See `grantpt(3)` in the UNIX
39/// manual pages for details.
40pub fn grantpt(master: &File) -> Result<()> {
41    // # Safety
42    // `grantpt()` is passed a valid file descriptor.
43    match unsafe { raw::grantpt(master.as_raw_fd()) } {
44        -1 => Err(Error::last_os_error()),
45        _ => Ok(()),
46    }
47}
48
49/// "Unlocks" the slave pty associated with the give nopen
50/// master. See `unlockpt(3)` in the UNIX manual pages for
51/// details.
52pub fn unlockpt(master: &File) -> Result<()> {
53    // # Safety
54    // `unlockpt()` will safely accept an invalid file descriptor.
55    match unsafe { raw::unlockpt(master.as_raw_fd()) } {
56        -1 => Err(Error::last_os_error()),
57        _ => Ok(()),
58    }
59}

61/// Returns the name of the slave pty associated with the
62/// given open master. See `unlockpt(3)` in the UNIX manual
63/// pages for details.
64pub fn ptsname(master: &File) -> Result<PathBuf> {
65    // # Safety
66    // `ptsname()` is passed a valid file descriptor.
67    let cstr = match unsafe { raw::ptsname(master.as_raw_fd()).as_ref() } {
68        None => return Err(Error::last_os_error()),
69        // # Safety
70        // `ptsname()` is guaranteed to return a valid pointer
71        // when it does not return an error.
72        Some(ptr) => unsafe { CStr::from_ptr(ptr) },
73    };
74    let buf = match cstr.to_str() {
75        Ok(s) => s,
76        Err(e) => return Err(Error::new(ErrorKind::InvalidData, e)),
77    };
78    let os_string = OsString::from_vec(buf.as_bytes().to_vec());
79    Ok(PathBuf::from(os_string))
80}

82/// Blocking wait until process completes. Returns the child
83/// process status. See `waitpid(3)` in the UNIX manual pages
84/// for details.
85pub fn waitpid(pid: i32) -> Result<ExitStatus> {
86    let mut status: c_int = 0;
87    // # Safety
88    // `waitpid()` will safely accept an invalid `pid`.
89    match unsafe { raw::waitpid(pid as pid_t, &mut status as *mut c_int, 0) } {
90        -1 => Err(Error::last_os_error()),
91        _ => Ok(ExitStatus::from_raw(status)),
92    }
93}

97    match unsafe { raw::close(fd) } {
98        -1 => Err(Error::last_os_error()),
99        _ => Ok(()),
100    }

103/// Make the underlying file of `dst` refer to the
104/// underlying file of `src`. If `dst` is open, it will
105/// be closed first. See `dup2(2)` in the UNIX manual
106/// pages for details.
107pub fn dup2(old: &File, new_fd: crate::StdFd) -> Result<()> {
108    // # Safety
109    // `dup2()` will safely accept invalid file descriptors.
110    match unsafe { raw::dup2(old.as_raw_fd(), new_fd.as_raw_fd()) } {
111        -1 => Err(Error::last_os_error()),
112        _ => Ok(()),
113    }
114}

116/// Make a new pipe. The 0-th side of the resulting array is
117/// the read side. The 1-th side is the write side. See
118/// `pipe(2)` in the UNIX manual pages for details.
119pub fn pipe() -> Result<[File; 2]> {
120    let mut pipefds: [RawFd; 2] = [0; 2];
121    // # Safety
122    // `pipe()` mutates only array to be returned, supplying
123    // it with valid values before returning success.
124    let result = unsafe { raw::pipe(pipefds.as_mut_ptr()) };
125    if result == -1 {
126        return Err(Error::last_os_error());
127    }
128    Ok(std::array::from_fn(|i| {
129        // # Safety
130        // On success `pipefds` will contain valid fds.
131        unsafe { FromRawFd::from_raw_fd(pipefds[i]) }
132    }))
