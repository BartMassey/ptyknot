6//! Child process and pty support.
7//!
8//! Start a child process running a specified action. The
9//! process may have a new pseudo-tty as its controlling
10//! terminal, and may have pipes to the master for some of
11//! its file descriptors.  The caller receives handles for
12//! all of this, along with the process ID of the child. The
13//! caller can then later wait for the child to exit by
14//! dropping its last reference.
15

29/// Represents a standard file descriptor to be overwritten
30/// in the child process.
31#[derive(Copy, Clone, Debug)]
32pub struct StdFd(RawFd);
33
34impl StdFd {
35    /// Get the underlying raw file descriptor.
36    pub fn as_raw_fd(&self) -> RawFd {
37        self.0
38    }
39}
40
41/// Stdin target for the pty-side interface.
42pub fn pty_stdin() -> StdFd {
43    StdFd(0)
44}
45
46/// Stdout target for the pty-side interface.
47pub fn pty_stdout() -> StdFd {
48    StdFd(1)
49}
50
51/// Stderr target for the pty-side interface.
52pub fn pty_stderr() -> StdFd {
53    StdFd(2)
54}

56/// Parent information about the child process.
57pub struct PtyKnot {
58    /// Child process ID.
59    pub pid: i32,
60}

62impl Drop for PtyKnot {
63    // When the `PtyKnot` is dropped, its child process is waited for.
64    fn drop(&mut self) {
65        let _ = pty::waitpid(self.pid);
66    }
67}

69/// Return the master side of a ready-to-operate pseudo-terminal.
70///
71/// # Example
72///
73/// ```
74/// let mut master = ptyknot::make_pty().expect("could not make pty");
75/// let slave_name = ptyknot::pty::ptsname(&mut master)
76///                  .expect("could not get slave name");
77/// println!("{}", slave_name.to_str()
78///                .expect("could not convert slave name"));
79/// ```
80pub fn make_pty() -> Result<File> {
81    let master = OpenOptions::new()
82        .read(true)
83        .write(true)
84        .open("/dev/ptmx")?;
85    pty::grantpt(&master)?;
86    pty::unlockpt(&master)?;
87    Ok(master)
88}

90/// Which direction a pipe runs.
91pub enum PipeDirection {
92    /// Master reads from pipe, slave writes.
93    MasterRead,
94    /// Master writes to pipe, slave reads.
95    MasterWrite,
96}
97
98/// Information needed during the pipe plumbing process.
99pub struct Plumbing {
100    master: File,
101    slave: File,
102    slave_target: StdFd,
103}
104
105impl Plumbing {
106    /// Create a new pipe running in the specified
107    /// direction, and remember the file descriptor of the
108    /// given file. This will later allow the slave to
109    /// attach `slave_target` to the other end of the pipe.
110    pub fn new(direction: PipeDirection, slave_target: StdFd) -> Result<Plumbing> {
111        let [pipefds0, pipefds1] = pty::pipe()?;
112        let (master, slave) = match direction {
113            PipeDirection::MasterWrite => (pipefds1, pipefds0),
114            PipeDirection::MasterRead => (pipefds0, pipefds1),
115        };
116        Ok(Plumbing {
117            master,
118            slave,
119            slave_target,
120        })
121    }
122
123    /// Implement the slave side of the plumbing by ensuring
124    /// that the slave end of the pipe is attached to the
125    /// previously-supplied file descriptor.
126    fn plumb_slave(&self) -> Result<()> {
127        pty::close(self.master.as_raw_fd())?;
128        pty::dup2(&self.slave, self.slave_target)
129    }
130
131    /// Extract the master side of the pipe for use by
132    /// the parent.
133    pub fn get_master(self) -> Result<File> {
134        Ok(self.master)
135    }
136}

138/// Start a child process running the given action,
139/// returning a `PtyKnot` for process information (currently
140/// just a process ID). When the the structure's destructor
141/// is called, it will wait to reap the child process and
142/// panic if it has crashed or exited with non-zero status.
143///
144/// The optional `pty` argument, if supplied with the master
145/// side of a pseudoterminal as created by `make_pty()`, will
146/// cause the child to be set up with the slave side of that
147/// pseudoterminal as its controlling terminal. Otherwise,
148/// the child will be set up with no controlling terminal.
149///
150/// The action must not access the PTY or plumbing values
151/// passed to this function. Their child-side descriptors are
152/// closed or reassigned before the action runs.
153///
154/// # Examples
155///
156/// ```
157/// use std::fs::OpenOptions;
158/// use std::io::{Write, BufRead, BufReader};
159///
160/// fn slave() {
161///     let mut tty = OpenOptions::new()
162///                   .write(true)
163///                   .open("/dev/tty")
164///                   .expect("cannot open /dev/tty");
165///     tty.write("hello world\n".as_bytes())
166///        .expect("cannot write to /dev/tty");
167///     tty.flush().expect("cannot flush /dev/tty");
168/// }
169///
170/// let mut pty = ptyknot::make_pty().expect("could not make pty");
171/// let knot = ptyknot::ptyknot(slave, Some(&mut pty), &vec![])
172///            .expect("cannot create slave");
173/// let mut tty = BufReader::new(&pty);
174/// let mut message = String::new();
175/// tty.read_line(&mut message)
176///    .expect("could not read message");
177/// // This will wait for the child.
178/// drop(knot);
179/// ```
180pub fn ptyknot<F: FnOnce()>(
181    action: F,
182    pty: Option<&File>,
183    plumbing: &[&Plumbing],
184) -> Result<PtyKnot> {
185    // # Safety
186    // `fork()` has no UB possibilities. It will
187    // either succeed or fail.
188    let pid = unsafe { libc::fork() };
189    match pid {
190        -1 => Err(Error::last_os_error()),
191        0 => {
192            let child_result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
193                // In the child process, there's no opportunity
194                // to return an error, so we'll just panic if
195                // there's a problem.
196
197                // Thanks much to
198                // https://www.win.tue.nl/~aeb/linux/lk/lk-10.html
199                // at "Getting a controlling tty" for helping
200                // understand this mess.
201
202                // Get rid of the current controlling terminal.
203                // # Safety
204                // `setsid()` has no UB possibilities. It will
205                // either succeed or fail.
206                if unsafe { libc::setsid() } == -1 {
207                    panic!(
208                        "could not lose controlling terminal: {}",
209                        Error::last_os_error()
210                    );
211                }
212
213                // Set a new controlling terminal if desired by
214                // opening a pty.
215                let mut slave = None;
216                if let Some(master) = pty {
217                    let slave_name = pty::ptsname(master).expect("cannot get pty name");
218                    pty::close(master.as_raw_fd()).expect("cannot close pty master");
219                    // Open the pty, which will set it
220                    // as the controlling terminal.
221                    let slave_fd = OpenOptions::new()
222                        .read(true)
223                        .write(true)
224                        .open(slave_name)
225                        .expect("cannot open pty");
226                    // Need to leave the slave pty open in case
227                    // the slave is going to open it, to avoid
228                    // a race.
229                    slave = Some(slave_fd);
230                }
231
232                // Set up any requested plumbing.
233                for p in plumbing {
234                    p.plumb_slave().expect("could not plumb pipe");
235                }
236
237                // Run the user action.
238                action();
239
240                // The slave side is no longer needed.
241                drop(slave);
242            }));
243
244            let status = if child_result.is_ok() { 0 } else { 101 };
245            // # Safety
246            // This is the child process and no Rust values
247            // may be dropped after their descriptors have
248            // been closed or reassigned.
249            unsafe { libc::_exit(status) }
250        }
251        _ => Ok(PtyKnot { pid }),
252    }
253}

255/// Provide a cleaner interface to `ptyknot()` *et al* by
256/// doing variable declaration and redeclaration.  The
257/// first argument is the identifier for the resulting knot.
258/// The second argument is the child action, as with
259/// `ptyknot()`. The rest of the arguments are:
260///
261/// * Zero or one pty redirections, consisting of `@`
262///   followed by a pseudo-tty identifier.
263/// * Zero or more master read redirections, consisting of
264///   `<` followed by a master read identifier and
265///   a standard file descriptor expression.
266/// * Zero or more master write redirections, consisting of
267///   `>` followed by a master write identifier and
268///   a standard file descriptor expression.
269///
270/// The macro will `let`-declare the necessary handles,
271/// assemble them and pass them to `ptyknot()`, then
272/// redeclare the handles to allow the master to manipulate
273/// them.
274///
275/// #Example
276///
277/// ```
278/// # use ptyknot::*;
279/// use std::fs::OpenOptions;
280/// use std::io::{Write, BufRead, BufReader};
281///
282/// fn slave() {
283///     let mut tty = OpenOptions::new()
284///                   .write(true)
285///                   .open("/dev/tty")
286///                   .expect("cannot open /dev/tty");
287///     tty.write("hello world\n".as_bytes())
288///        .expect("cannot write to /dev/tty");
289///     tty.flush().expect("cannot flush /dev/tty");
290/// }
291///
292/// # pub fn main() {
293/// ptyknot!(knot, slave, @ pty);
294/// let mut tty = BufReader::new(&pty);
295/// let mut message = String::new();
296/// tty.read_line(&mut message)
297///    .expect("could not read message");
298/// // This will wait for the child.
299/// drop(knot);
300/// # }
301/// ```
302#[macro_export]
303macro_rules! ptyknot {
304    ($knot:ident,
305     $slave:expr
306     $(, @ $tty:ident)*
307     $(, < $master_read:ident $read_fd:expr)*
308     $(, > $master_write:ident $write_fd:expr)*) => {
309        $(let mut $tty = $crate::make_pty().expect("could not make pty");)*
310        $(let $master_read =
311          $crate::Plumbing::new($crate::PipeDirection::MasterRead,$read_fd)
312          .expect("$master_read: create failed");)*
313        $(let $master_write =
314          $crate::Plumbing::new(PipeDirection::MasterWrite,$write_fd)
315          .expect("$master_write: create failed");)*
316        let $knot =
317            $crate::ptyknot($slave,
318                            match [$(&$tty)*].len() {
319                                0 => None,
320                                _ => Some(&mut $($tty)*),
321                            },
322                            &[$(&$master_read,)* $(&$master_write,)*])
323            .expect("ptyknot failed");
324        $(let $master_read =
325          $master_read.get_master()
326          .expect("$master_read: get master failed");)*
327        $(let mut $master_write =
328          $master_write.get_master()
329          .expect("$master_write: get master failed");)*
330    }
331}

333#[cfg(test)]
334fn pty_slave() {
335    let mut tty = OpenOptions::new()
336        .write(true)
337        .open("/dev/tty")
338        .expect("cannot open /dev/tty");
339    tty.write_all("hello world\n".as_bytes())
340        .expect("cannot write to /dev/tty");
341    tty.flush().expect("cannot flush /dev/tty");
342}

344#[test]
345fn pty_test() {
346    let pty = make_pty().expect("could not make pty");
347    let knot = ptyknot(pty_slave, Some(&pty), &[]).expect("ptyknot fail");
348    let mut master = BufReader::new(&pty);
349    let mut message = String::new();
350    master
351        .read_line(&mut message)
352        .expect("could not read message");
353    drop(knot);
354    assert_eq!(message.trim(), "hello world");
355}

404#[test]
405fn plumbing_preserves_parent_standard_descriptors() {
406    let descriptors = [pty_stdin(), pty_stdout(), pty_stderr()];
407    for descriptor in descriptors {
408        let fd_path = format!("/proc/self/fd/{}", descriptor.as_raw_fd());
409        assert!(
410            std::fs::metadata(&fd_path).is_ok(),
411            "standard descriptor was not open before plumbing: {fd_path}"
412        );
413
414        let plumbing = Plumbing::new(PipeDirection::MasterRead, descriptor)
415            .expect("could not create plumbing");
416        drop(plumbing.get_master().expect("could not get master"));
417
418        assert!(
419            std::fs::metadata(&fd_path).is_ok(),
420            "plumbing closed parent standard descriptor: {fd_path}"
421        );
422    }
